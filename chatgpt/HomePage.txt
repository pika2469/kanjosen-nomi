import { useEffect, useMemo, useRef, useState } from 'react'
import { useGameStore } from '@/store/gameStore'
import  { Settings as SettingsIcon } from 'lucide-react'

type FloatItem = {
  key: string
  src: string
  alt: string
  base: { x: number; y: number } // px（親の左上基準）
  size: number // px
  opacity: number
  // ランダム挙動パラメータ
  dur: number // sec
  delay: number // sec
  driftX: number // px
  driftY: number // px
  rot: number // deg
  scaleAmp: number // ★ 追加：スケール振れ幅（0.03〜0.07など）
}

function rand(min: number, max: number) {
  return min + Math.random() * (max - min)
}

export function HomePage() {
  const { setPage, players } = useGameStore()
  const canStart = players.length >= 2

  // 任意：パララックス（視差）
  const wrapRef = useRef<HTMLDivElement | null>(null)
  const [parallax, setParallax] = useState({ x: 0, y: 0 })

  // Home入場（CTAを軽く出す）
  const [entered, setEntered] = useState(false)

  // motion抑制設定
  const reduceMotion = useMemo(() => {
    const m = window.matchMedia?.('(prefers-reduced-motion: reduce)')
    return !!m?.matches
  }, [])

  // 漂うアイコン：ランダムパラメータ（初回マウント時に固定）
  const floatItems = useMemo<FloatItem[]>(() => {
    // 既存の配置を踏襲しつつ、挙動だけランダム化
    const base = [
      {
        key: 'train',
        src: '/home/float/train.svg',
        alt: '',
        base: { x: 32, y: 96 }, // left-8 top-24
        size: 80, // w-20
        opacity: 0.75,
      },
      {
        key: 'dice',
        src: '/home/float/dice.svg',
        alt: '',
        base: { x: 9999, y: 192 }, // right-12 top-48（右基準は後でstyleで扱う）
        size: 80,
        opacity: 0.75,
        // 右寄せフラグ代わり：x=9999
      },
      {
        key: 'beer',
        src: '/home/float/beer.svg',
        alt: '',
        base: { x: 48, y: 9999 }, // left-12 bottom-28（下基準は後でstyleで扱う）
        size: 80,
        opacity: 0.7,
      },
      {
        key: 'map',
        src: '/home/float/map.svg',
        alt: '',
        base: { x: 9999, y: 9999 }, // right-10 bottom-52
        size: 80,
        opacity: 0.7,
      },
    ]

    return base.map((b) => ({
      ...b,
      dur: rand(8.0, 12.0), // ★ 少し幅を広げて同期感を減らす
      delay: rand(-3.2, 0.0), // ★ 個体差：負のdelayで「最初から動いている」感じ
      driftX: rand(-26, 26), // ★ 左右の行動範囲を拡大（CSS内の動きと合算される）
      driftY: rand(-22, 22), // ★ 上下の行動範囲も拡大
      rot: rand(-7, 7),
      scaleAmp: rand(0.03, 0.07), // ★ 大きさ変化（3〜7%）
    }))
  }, [])

  useEffect(() => {
    // Home入場（MainLayoutのpage-enter対象外のため）
    const t = window.setTimeout(() => setEntered(true), 30)
    return () => window.clearTimeout(t)
  }, [])

  useEffect(() => {
    const el = wrapRef.current
    if (!el) return
    if (reduceMotion) return

    const onMove = (e: MouseEvent) => {
      const r = el.getBoundingClientRect()
      const cx = r.left + r.width / 2
      const cy = r.top + r.height / 2
      const dx = (e.clientX - cx) / r.width
      const dy = (e.clientY - cy) / r.height
      setParallax({ x: dx, y: dy })
    }

    window.addEventListener('mousemove', onMove, { passive: true })
    return () => window.removeEventListener('mousemove', onMove)
  }, [reduceMotion])

  // レイヤーごとの視差量（小さめが上品）
  const bgTx = parallax.x * 10
  const bgTy = parallax.y * 10
  const iconTx = parallax.x * 18
  const iconTy = parallax.y * 18

  return (
    <div ref={wrapRef} className="relative h-full w-full bg-transparent overflow-hidden">
      {/* 背景パターン（Homeのみ） */}
      <div className="pointer-events-none fixed inset-0 z-0 overflow-hidden">
        <div className="absolute left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2">
          <img
            src="/home/bg/loop_pattern.svg"
            alt=""
            draggable={false}
            className="
              w-[120vmax] h-[110vmax]
              max-w-[1400px] max-h-[1400px]
              opacity-[0.28]
              animate-homeBgSpin
              select-none
            "
            style={{ transform: `translate3d(${bgTx}px, ${bgTy}px, 0)` }}
          />
        </div>

        {/* うっすら光 */}
        <div className="absolute -top-24 left-1/2 h-64 w-64 -translate-x-1/2 rounded-full bg-sky-200/25 blur-3xl" />
        <div className="absolute -bottom-28 right-[-40px] h-72 w-72 rounded-full bg-indigo-200/20 blur-3xl" />
      </div>

      {/* 右上：≡（設定） */}
      <button
        type="button"
        onClick={() => setPage('settings')}
        className="
          fixed top-6 right-5 z-30
          rounded-2xl bg-white/65 px-3 py-2
          shadow-md ring-1 ring-black/5 backdrop-blur
          transition
          hover:scale-110 hover:bg-white/85 hover:shadow-lg
          active:scale-95
        "
        aria-label="設定"
      >
        <SettingsIcon size={20} className="text-slate-800" />
        {/* <span className="block text-lg leading-none text-slate-800">≡</span> */}
      </button>

      {/* 漂うアイコン（ランダム化 + delay分散 + スケール変化） */}
        
      <div
        className="pointer-events-none absolute inset-0"
        style={{ transform: `translate3d(${iconTx}px, ${iconTy}px, 0)` }}
      >
        {floatItems.map((it) => {
          const isRight = it.base.x === 9999
          const isBottom = it.base.y === 9999

          const style: React.CSSProperties = {
            width: it.size,
            opacity: it.opacity,
            animationDuration: `${it.dur}s`,
            animationDelay: `${it.delay}s`, // ★ 個体差delay
            // 個体差（CSS変数で取り込む：ドリフト/回転/スケール）
            ['--fx' as any]: `${it.driftX}px`,
            ['--fy' as any]: `${it.driftY}px`,
            ['--fr' as any]: `${it.rot}deg`,
            ['--fs' as any]: `${it.scaleAmp}`, // ★ 追加：スケール振れ幅
            ...(isRight
              ? { right: it.key === 'dice' ? '3rem' : '2.5rem' }
              : { left: `${it.base.x}px` }),
            ...(isBottom
              ? { bottom: it.key === 'beer' ? '7rem' : '13rem' }
              : { top: `${it.base.y}px` }),
          }

          return (
            <img
              key={it.key}
              src={it.src}
              alt={it.alt}
              draggable={false}
              className={[
                'absolute drop-shadow-sm select-none',
                reduceMotion
                  ? ''
                  : it.key === 'dice' || it.key === 'map'
                    ? 'ks-homeFloat2'
                    : 'ks-homeFloat',
              ].join(' ')}
              style={style}
            />
          )
        })}
      </div>

      {/* 中央：2ボタン */}
      <div className="relative flex h-full w-full items-center justify-center">
        <section className="w-full max-w-sm bg-transparent">
          <div
            className={[
              'space-y-4 -translate-y-10 bg-transparent',
              'transition-all duration-300 ease-out',
              entered ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-2',
            ].join(' ')}
          >
            {/* ゲーム開始（リッチCTA） */}
            <button
              type="button"
              disabled={!canStart}
              onClick={() => setPage('mood')}
              className={[
                'group relative w-full overflow-hidden rounded-2xl py-6 text-base font-semibold',
                'shadow-none transition-all duration-200',
                canStart
                  ? 'hover:-translate-y-1 hover:shadow-[0_16px_36px_rgba(2,6,23,0.20)] active:translate-y-0 active:scale-[0.99]'
                  : 'cursor-not-allowed opacity-55',
              ].join(' ')}
            >
              <span className="absolute inset-0 bg-gradient-to-r from-slate-950 via-slate-900 to-slate-800" />
              <span className="absolute inset-0 bg-gradient-to-tr from-white/25 via-white/5 to-transparent" />
              <span className="absolute -left-1/2 top-0 h-full w-1/2 rotate-12 bg-white/15 blur-xl transition-transform duration-500 group-hover:translate-x-[260%]" />
              {/* 呼吸するグロー（canStart時のみ） */}
              {canStart && (
                <span className="absolute -inset-6 rounded-[28px] ks-cta-glow" />
              )}

              <span className="relative inline-flex items-center justify-center gap-2 text-white">
                <span className="text-lg transition-transform duration-200 group-hover:-translate-y-0.5 group-hover:rotate-[-6deg] group-hover:scale-110">
                  🍺
                </span>
                ゲーム開始
              </span>
            </button>

            {!canStart && (
              <div className="rounded-2xl bg-amber-50/90 px-4 py-3 text-xs text-amber-900 ring-1 ring-amber-200 backdrop-blur">
                ゲーム開始には2人以上必要です。<br />
                右上の「≡」→設定→プレイヤー設定から追加できます。
              </div>
            )}

            {/* ミニゲーム */}
            <button
              type="button"
              onClick={() => setPage('minigame')}
              className="
                group w-full rounded-2xl
                border border-white/60 bg-white/55 py-6 text-base font-semibold text-slate-900
                shadow-none ring-1 ring-black/5
                transition-all duration-200
                hover:-translate-y-1 hover:bg-white/20 hover:shadow-[0_18px_40px_rgba(2,6,23,0.16)]
                active:translate-y-0 active:scale-[0.99]
              "
            >
              <span className="inline-flex items-center justify-center gap-2">
                <span className="text-lg transition-transform duration-200 group-hover:-translate-y-0.5 group-hover:rotate-[6deg] group-hover:scale-110">
                  🎮
                </span>
                ミニゲームへ
              </span>
            </button>
          </div>
        </section>
      </div>

      {/* Home専用アニメ（MainLayoutに依存しない） */}
      <style>{`
        @keyframes homeBgSpin {
          0% { transform: rotate(0deg) scale(1); }
          100% { transform: rotate(360deg) scale(1); }
        }
        .animate-homeBgSpin {
          transform-origin: 50% 50%;
          animation: homeBgSpin 36s linear infinite;
        }

        /* ランダム値（CSS変数）を取り込んで微妙に個体差を出す
           - --fx/--fy: 追加ドリフト（左右/上下）
           - --fr: 回転差
           - --fs: スケール振れ幅
        */
        @keyframes ks-homeFloat {
          0% {
            transform: translate3d(0px, 0px, 0) rotate(0deg) scale(1);
          }
          25% {
            transform: translate3d(calc(18px + var(--fx, 0px)), calc(-14px + var(--fy, 0px)), 0)
              rotate(calc(-2deg + var(--fr, 0deg)))
              scale(calc(1 + var(--fs, 0.04)));
          }
          50% {
            transform: translate3d(calc(-34px + var(--fx, 0px)), calc(-30px + var(--fy, 0px)), 0)
              rotate(calc(-4deg + var(--fr, 0deg)))
              scale(1);
          }
          75% {
            transform: translate3d(calc(24px + var(--fx, 0px)), calc(-18px + var(--fy, 0px)), 0)
              rotate(calc(2deg + var(--fr, 0deg)))
              scale(calc(1 - var(--fs, 0.04)));
          }
          100% {
            transform: translate3d(0px, 0px, 0) rotate(0deg) scale(1);
          }
        }

        @keyframes ks-homeFloat2 {
          0% {
            transform: translate3d(0px, 0px, 0) rotate(0deg) scale(1);
          }
          25% {
            transform: translate3d(calc(-26px + var(--fx, 0px)), calc(-16px + var(--fy, 0px)), 0)
              rotate(calc(3deg + var(--fr, 0deg)))
              scale(calc(1 - var(--fs, 0.04)));
          }
          50% {
            transform: translate3d(calc(40px + var(--fx, 0px)), calc(-34px + var(--fy, 0px)), 0)
              rotate(calc(6deg + var(--fr, 0deg)))
              scale(1);
          }
          75% {
            transform: translate3d(calc(-20px + var(--fx, 0px)), calc(-22px + var(--fy, 0px)), 0)
              rotate(calc(3deg + var(--fr, 0deg)))
              scale(calc(1 + var(--fs, 0.04)));
          }
          100% {
            transform: translate3d(0px, 0px, 0) rotate(0deg) scale(1);
          }
        }

        .ks-homeFloat { animation: ks-homeFloat 8.0s ease-in-out infinite; }
        .ks-homeFloat2 { animation: ks-homeFloat2 9.5s ease-in-out infinite; }

        @keyframes ksCtaGlow {
          0%, 100% { opacity: 0.22; transform: scale(1); }
          50%      { opacity: 0.48; transform: scale(1.03); }
        }

        .ks-cta-glow {
          background: radial-gradient(circle at 30% 20%, rgba(255,255,255,0.55), rgba(255,255,255,0.10) 55%, transparent 72%),
                      radial-gradient(circle at 70% 60%, rgba(56,189,248,0.22), transparent 60%);
          filter: blur(14px);
          animation: ksCtaGlow 2.6s ease-in-out infinite;
        }

        @media (prefers-reduced-motion: reduce) {
          .animate-homeBgSpin, .ks-homeFloat, .ks-homeFloat2, .ks-cta-glow { animation: none; }
        }
      `}</style>
    </div>
  )
}
