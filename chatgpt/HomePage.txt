import { useEffect, useRef, useState } from 'react'
import { useGameStore } from '@/store/gameStore'

export function HomePage() {
  const { setPage, players } = useGameStore()
  const canStart = players.length >= 2

  // 任意：パララックス（視差）
  const wrapRef = useRef<HTMLDivElement | null>(null)
  const [parallax, setParallax] = useState({ x: 0, y: 0 })

  useEffect(() => {
    const el = wrapRef.current
    if (!el) return

    // motion抑制設定なら無効化
    const m = window.matchMedia?.('(prefers-reduced-motion: reduce)')
    if (m?.matches) return

    const onMove = (e: MouseEvent) => {
      const r = el.getBoundingClientRect()
      const cx = r.left + r.width / 2
      const cy = r.top + r.height / 2
      const dx = (e.clientX - cx) / r.width
      const dy = (e.clientY - cy) / r.height
      setParallax({ x: dx, y: dy })
    }

    window.addEventListener('mousemove', onMove, { passive: true })
    return () => window.removeEventListener('mousemove', onMove)
  }, [])

  // レイヤーごとの視差量（小さめが上品）
  const bgTx = parallax.x * 10
  const bgTy = parallax.y * 10
  const iconTx = parallax.x * 18
  const iconTy = parallax.y * 18

  return (
    <div ref={wrapRef} className="relative h-full w-full bg-transparent overflow-hidden">
      {/* 背景パターン（Homeのみ） */}
      <div className="pointer-events-none fixed inset-0 z-0 overflow-hidden">
        <div className="absolute left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2">
            {/* 回転用 */}
            <img
                src="/home/bg/loop_pattern.svg"
                alt=""
                draggable={false}
                className="
                    w-[120vmax] h-[110vmax]
                    max-w-[1400px] max-h-[1400px]
                    opacity-[0.28]
                    animate-homeBgSpin
                    select-none
                "
                style={{ transform: `translate3d(${bgTx}px, ${bgTy}px, 0)` }}
            />
        </div>
            

        {/* うっすら光（MainLayout背景との継ぎ目を馴染ませる） */}
        <div className="absolute -top-24 left-1/2 h-64 w-64 -translate-x-1/2 rounded-full bg-sky-200/25 blur-3xl" />
        <div className="absolute -bottom-28 right-[-40px] h-72 w-72 rounded-full bg-indigo-200/20 blur-3xl" />
      </div>

      {/* 右上：≡（設定） */}
      <button
        type="button"
        onClick={() => setPage('settings')}
        className="
          fixed top-6 right-5 z-30
          rounded-2xl bg-white/65 px-3 py-2
          shadow-md ring-1 ring-black/5 backdrop-blur
          transition
          hover:scale-110 hover:bg-white/85 hover:shadow-lg
          active:scale-95
        "
        aria-label="設定"
      >
        <span className="block text-lg leading-none text-slate-800">≡</span>
      </button>

      {/* 漂うアイコン（絵文字→SVG画像） */}
      <div
        className="pointer-events-none absolute inset-0"
        style={{ transform: `translate3d(${iconTx}px, ${iconTy}px, 0)` }}
      >
        <img
          src="/home/float/train.svg"
          alt=""
          className="floaty absolute left-8 top-24 w-20 opacity-75 drop-shadow-sm"
        />
        <img
          src="/home/float/dice.svg"
          alt=""
          className="floaty2 absolute right-12 top-48 w-20 opacity-75 drop-shadow-sm"
        />
        <img
          src="/home/float/beer.svg"
          alt=""
          className="floaty absolute left-12 bottom-28 w-20 opacity-70 drop-shadow-sm"
        />
        <img
          src="/home/float/map.svg"
          alt=""
          className="floaty2 absolute right-10 bottom-52 w-20 opacity-70 drop-shadow-sm"
        />
      </div>

      {/* 中央：2ボタン（あなたの現行を踏襲） */}
      <div className="relative flex h-full w-full items-center justify-center">
        <section className="w-full max-w-sm bg-transparent">
          <div className="space-y-4 -translate-y-10 bg-transparent">
            {/* ゲーム開始（リッチCTA） */}
            <button
              type="button"
              disabled={!canStart}
              onClick={() => setPage('mood')}
              className={[
                'group relative w-full overflow-hidden rounded-2xl py-6 text-base font-semibold',
                'shadow-none transition-all duration-200',
                canStart
                  ? 'hover:-translate-y-1 hover:shadow-[0_16px_36px_rgba(2,6,23,0.20)] active:translate-y-0 active:scale-[0.99]'
                  : 'cursor-not-allowed opacity-55',
              ].join(' ')}
            >
              <span className="absolute inset-0 bg-gradient-to-r from-slate-950 via-slate-900 to-slate-800" />
              <span className="absolute inset-0 bg-gradient-to-tr from-white/25 via-white/5 to-transparent" />
              <span className="absolute -left-1/2 top-0 h-full w-1/2 rotate-12 bg-white/15 blur-xl transition-transform duration-500 group-hover:translate-x-[260%]" />
              <span className="relative inline-flex items-center justify-center gap-2 text-white">
                <span className="text-lg transition-transform duration-200 group-hover:-translate-y-0.5 group-hover:rotate-[-6deg] group-hover:scale-110">
                  🍺
                </span>
                ゲーム開始
              </span>
            </button>

            {!canStart && (
              <div className="rounded-2xl bg-amber-50/90 px-4 py-3 text-xs text-amber-900 ring-1 ring-amber-200 backdrop-blur">
                ゲーム開始には2人以上必要です。<br />
                右上の「≡」→設定→プレイヤー設定から追加できます。
              </div>
            )}

            {/* ミニゲーム */}
            <button
              type="button"
              onClick={() => setPage('minigame')}
              className="
                group w-full rounded-2xl
                border border-white/60 bg-white/55 py-6 text-base font-semibold text-slate-900
                shadow-none ring-1 ring-black/5
                transition-all duration-200
                hover:-translate-y-1 hover:bg-white/20 hover:shadow-[0_18px_40px_rgba(2,6,23,0.16)]
                active:translate-y-0 active:scale-[0.99]
              "
            >
              <span className="inline-flex items-center justify-center gap-2">
                <span className="text-lg transition-transform duration-200 group-hover:-translate-y-0.5 group-hover:rotate-[6deg] group-hover:scale-110">
                  🎮
                </span>
                ミニゲームへ
              </span>
            </button>
          </div>
        </section>
      </div>

      {/* Home専用アニメ（MainLayoutに依存しない） */}
      <style>{`
        @keyframes homeBg {
          0% { background-position: 0px 0px; }
          100% { background-position: 240px 320px; }
        }
        .animate-homeBg { animation: homeBg 18s linear infinite; }

        @keyframes homeBgSpin {
            0% { transform: rotate(0deg) scale(1); }
            100% { transform: rotate(360deg) scale(1); }
        }
        .animate-homeBgSpin {
            transform-origin: 50% 50%;
            animation: homeBgSpin 36s linear infinite;
        }

        @keyframes floaty {
          0% { transform: translateY(0px) rotate(0deg); }
          50% { transform: translateY(-10px) rotate(-2deg); }
          100% { transform: translateY(0px) rotate(0deg); }
        }
        @keyframes floaty2 {
          0% { transform: translateY(0px) rotate(0deg); }
          50% { transform: translateY(-14px) rotate(3deg); }
          100% { transform: translateY(0px) rotate(0deg); }
        }
        .floaty { animation: floaty 4.8s ease-in-out infinite; }
        .floaty2 { animation: floaty2 5.6s ease-in-out infinite; }

        @media (prefers-reduced-motion: reduce) {
          .animate-homeBgSpin, .floaty, .floaty2 { animation: none; }
        }
      `}</style>
    </div>
  )
}
