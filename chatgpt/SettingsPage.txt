import { Button } from '@/components/ui/button'
import { useGameStore } from '@/store/gameStore'
import { STATIONS } from '@/stations'
import type { StationId } from '@/types/game'

export function SettingsPage() {
    const { settings, setSettings, resetAllData, players, setPlayerMaxDrink } = useGameStore()

    return (
        <div className="space-y-4">
            <section>
                <h2 className="text-xl font-bold mb-1">設定</h2>
                <p className="text-sm text-gray-600">
                    各種設定を行うページ
                </p>
            </section>

            <section className="space-y-3 text-sm">
                <div className="flex items-center justify-between rounded-lg border bg-white p-3">
                    <div className="flex flex-col text-left">
                        <div className="font-medium">駅重複</div>
                        <div className="text-xs text-gray-500">同じ駅を複数回訪問することを許可します。</div>
                    </div>
                    <Button
                        size="sm"
                        variant={settings.allowDuplicateStations ? 'default' : 'outline'}
                        onClick={() => setSettings({ allowDuplicateStations: !settings.allowDuplicateStations })}
                    >
                        {settings.allowDuplicateStations ? 'あり' : 'なし'}
                    </Button>
                </div>

                <div className="flex items-center justify-between rounded-lg border bg-white p-3">
                    <div className="flex flex-col text-left">
                        <div className="font-medium">開始駅設定</div>
                        <div className="text-xs text-gray-500">ゲーム開始時に規準となる駅を選択します</div>
                    </div>
                    
                    <select
                        className="rounded border px-2 py-1 text-sm"
                        value={settings.startStationId}
                        onChange={(e) => setSettings({ startStationId: e.target.value as StationId })}
                    >
                        {STATIONS.map((station) => (
                            <option key={station.id} value={station.id}>
                                {station.name}
                            </option>
                        ))}
                    </select>
                </div>

                <div className="flex items-center justify-between rounded-lg border bg-white p-3">
                    <div className="flex flex-col text-left">
                        <div className="font-medium">サウンド</div>
                        <div className="text-xs text-gray-500">BGMや効果音を有効にします</div>
                    </div>
                    <Button
                        size="sm"
                        variant={settings.sound ? 'default' : 'outline'}
                        onClick={() => setSettings({ sound: !settings.sound })}
                    >
                        {settings.sound ? 'ON' : 'OFF'}
                    </Button>
                </div>

                <div className="flex items-center justify-between rounded-lg border bg-white p-3">
                    <div className="flex flex-col text-left">
                        <div className="font-medium">セーフティモード</div>
                        <div className="text-xs text-gray-500">飲酒杯数を減らします</div>
                    </div>
                    <Button
                        size="sm"
                        variant={settings.safety ? 'default' : 'outline'}
                        onClick={() => setSettings({ safety: !settings.safety })}
                    >
                        {settings.safety ? 'ON' : 'OFF'}
                    </Button>
                </div>

                <div className="mt-2 rounded-lg border bg-white p-3">
                    <div className="mb-2 flex flex-col text-left">
                        <div className="font-medium">プレイヤー別 最大杯数</div>
                        <div className="text-xs text-gray-500">
                            各プレイヤーの最大杯数を設定します
                        </div>
                    </div>

                    {players.length === 0 ? (
                        <p className="text-xs text-gray-400">プレイヤーが登録されていません</p>
                    ) : (
                        <div className="space-y-2">
                            {players.map((p) => (
                                <div key={p.id} className="flex items-center justify-between gap-2">
                                    <span className="text-xs">{p.name}</span>
                                    <select
                                        className="rounded border px-2 py-1 text-xs"
                                        value={p.maxDrink ?? 5}
                                        onChange={(e) => setPlayerMaxDrink(p.id, Number(e.target.value))}
                                    >
                                        {[3, 4, 5, 6].map((v) => (
                                            <option key={v} value={v}>
                                                最大 {v} 杯
                                            </option>
                                        ))}
                                    </select>
                                </div>
                            ))}
                        </div>
                    )}
                </div>
            </section>

            <section className="pt-2">
                <Button variant="destructive" className="w-full" onClick={resetAllData}>
                    データリセット
                </Button>
                <p className="mt-1 text-xs text-gray-500">
                    プレイヤー情報・設定をすべてリセットします。
                </p>
            </section>
        </div>
    )
}
